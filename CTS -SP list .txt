-- =============================================      
-- Author  : Thiras    
-- Create date : 07/11/2013    
-- Description : Retrieving WIP volumes and issues    
-- =============================================    
-- Prasanta | 27 Nov 2013 | updated to upper case

--  [usp_PTS_GetWIPArticles] 49,4821  
ALTER PROCEDURE [dbo].[usp_PTS_GetWIPArticles]      
	@customerCode int=0,      
	@journalCode int=0 ,     
	@Volume varchar(20) = '0',
	@Issue varchar(20) = '0'
AS      
BEGIN      
	SET NOCOUNT ON;      
	DECLARE @JOBID INT
	
	SELECT  @JOBID = JOBID FROM JOURNAL WITH(NOLOCK) WHERE Jour_ccode=@CUSTOMERCODE 
	AND Jour_code=@JOURNALCODE 
	
	if exists( SELECT  1    
	FROM ARTICLE A WITH(NOLOCK)  WHERE A.Art_Ccode=@CUSTOMERCODE 
	AND A.Art_Jcode=@JOURNALCODE 
	AND A.JobChapterID IN (   SELECT JobChapterId FROM TBL_JOB_CHAPSTAGEDETAILS R   WITH(NOLOCK)  
	where JobId=@JOBID AND (R.STAGERECEIVEDDATE <> '1900/01/01' AND R.STAGERECEIVEDDATE IS NOT NULL) 
	AND (R.DISPATCHDATE = '1900/01/01' OR R.DISPATCHDATE IS NULL )     )  
	 
	and (  @Volume  = '0' or Art_volno = RTRIM(@Volume) ) and (@Issue = '0' or Art_issno = @Issue))
	begin
		 SELECT  UPPER(RTRIM(LTRIM(ART_ACODE))) AS FIELDID ,UPPER(RTRIM(LTRIM(JobChapterID))) AS FIELDNAME     
		 FROM ARTICLE A   WITH(NOLOCK)   
		 WHERE A.Art_Ccode=@CUSTOMERCODE AND A.JobChapterID IN (   SELECT JobChapterId FROM TBL_JOB_CHAPSTAGEDETAILS R  WITH(NOLOCK)
		 WHERE JobId=@JOBID AND (R.STAGERECEIVEDDATE <> '1900/01/01' AND R.STAGERECEIVEDDATE IS NOT NULL) AND (R.DISPATCHDATE = '1900/01/01' OR R.DISPATCHDATE IS NULL )     )  
		 AND A.Art_Jcode=@JOURNALCODE  
		 and (  @Volume  = '0' or Art_volno = RTRIM(@Volume) ) and (@Issue = '0' or Art_issno = @Issue)-- and isnull(TypeSetterName,'')!='MPS'
		 order by FIELDID 
	end
	else
	begin
		SELECT  UPPER(RTRIM(LTRIM(ART_ACODE))) AS FIELDID ,UPPER(RTRIM(LTRIM(JobChapterID))) AS FIELDNAME     
		FROM ARTICLE A  WITH(NOLOCK)   
		WHERE A.Art_Ccode=@CUSTOMERCODE AND A.JobChapterID IN (   SELECT JobChapterId FROM TBL_JOB_CHAPSTAGEDETAILS R WITH(NOLOCK) 
		WHERE JobId=@JOBID AND (R.STAGERECEIVEDDATE <> '1900/01/01' AND R.STAGERECEIVEDDATE IS NOT NULL) AND R.SubStageId = 23  )  
		AND A.Art_Jcode=@JOURNALCODE  
		and (  @Volume  = '0' or Art_volno = RTRIM(@Volume) ) and (@Issue = '0' or Art_issno = @Issue)-- and isnull(TypeSetterName,'')!='MPS'
		order by FIELDID 
	end

	--AND R.SubStageId = @SUBSTAGEID [Ram]    

END
